const root = dv.container;
root.classList.add("university-dashboard");

const MONTHS_FULL = [
  "января", "февраля", "марта", "апреля", "мая", "июня",
  "июля", "августа", "сентября", "октября", "ноября", "декабря"
];

function plural(number, forms) {
  const abs = Math.abs(number) % 100;
  const last = abs % 10;
  if (abs > 10 && abs < 20) return forms[2];
  if (last > 1 && last < 5) return forms[1];
  if (last === 1) return forms[0];
  return forms[2];
}

function toDate(value) {
  if (!value) return null;
  if (typeof value === "object" && typeof value.toFormat === "function") return value;
  const parsed = DateTime.fromISO(String(value));
  return parsed.isValid ? parsed : null;
}

function openPath(path) {
  if (!path) return;
  app.workspace.openLinkText(path, "", false);
}

function makeInteractive(element, handler, label) {
  element.setAttribute("role", "button");
  element.setAttribute("tabindex", "0");
  if (label) element.setAttribute("aria-label", label);
  element.addEventListener("click", handler);
  element.addEventListener("keydown", event => {
    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      handler(event);
    }
  });
}

function isContentPage(page, excludedNames = []) {
  const name = page.file?.name || "";
  return (
    name &&
    !name.startsWith("_Index_of_") &&
    !name.startsWith("._") &&
    !name.startsWith("MOC") &&
    !excludedNames.includes(name)
  );
}

const page = root.createEl("div", { cls: "ud-page" });
const pages = dv.pages('"10-Университет"')
  .where(page => isContentPage(page, ["Университет"]))
  .array()
  .sort((a, b) => a.file.name.localeCompare(b.file.name, "ru"));

const groupsMap = new Map();
for (const item of pages) {
  const rel = (item.file.folder || "").replace("10-Университет/", "");
  const group = rel.split("/")[0] || "Разное";
  if (!groupsMap.has(group)) groupsMap.set(group, []);
  groupsMap.get(group).push(item);
}

const groups = [...groupsMap.entries()].sort((a, b) => a[0].localeCompare(b[0], "ru"));
const activeCount = pages.filter(item => item.status && item.status !== "Готово").length;
const deadlineCount = pages.filter(item => item.due_date && item.status !== "Готово").length;

const hero = page.createEl("section", { cls: "ud-note-hero" });
hero.createEl("div", { cls: "ud-note-eyebrow", text: "Автонавигация по учёбе" });
hero.createEl("h1", { cls: "ud-note-title", text: "Университет" });
hero.createEl("p", { cls: "ud-note-sub", text: "Все университетские заметки собраны по предметам и подкатегориям. Интерфейс упрощён под стабильный просмотр в Obsidian." });

const statGrid = hero.createEl("div", { cls: "ud-stat-grid" });
[
  ["Заметки", pages.length, `${groups.length} ${plural(groups.length, ["раздел", "раздела", "разделов"])}`],
  ["Активные", activeCount, activeCount ? "Есть элементы в работе" : "Всё спокойно"],
  ["С дедлайном", deadlineCount, deadlineCount ? "Есть сроки выполнения" : "Без сроков"],
  ["Группы", groups.length, groups.length ? "Автонавигация по папкам" : "Пока пусто"]
].forEach(([label, value, meta]) => {
  const card = statGrid.createEl("div", { cls: "ud-stat-card" });
  card.createEl("div", { cls: "ud-stat-label", text: String(label) });
  card.createEl("div", { cls: "ud-stat-value", text: String(value) });
  card.createEl("div", { cls: "ud-stat-meta", text: String(meta) });
});

const searchWrap = page.createEl("div", { cls: "ud-search-wrap" });
searchWrap.createEl("span", { cls: "ud-icon ud-search-icon", text: "search" });
const searchInput = searchWrap.createEl("input", {
  cls: "ud-search ud-focus-ring",
  attr: {
    type: "text",
    placeholder: "Поиск по заметкам раздела...",
    "aria-label": "Поиск по заметкам раздела"
  }
});

const renderedGroups = [];

function renderCard(parent, item) {
  const card = parent.createEl("div", { cls: "ud-card ud-searchable ud-focus-ring" });
  card.style.setProperty("--ud-border-color", "var(--ud-accent-action)");
  makeInteractive(card, () => openPath(item.file.path), `Открыть ${item.file.name}`);

  const main = card.createEl("div", { cls: "ud-card-main" });
  main.createEl("div", { cls: "ud-card-link ud-static-title", text: item.file.name });
  const meta = main.createEl("div", { cls: "ud-card-meta" });

  if (item.type) meta.createEl("span", { cls: "ud-card-tag", text: item.type });
  if (item.status && item.status !== "Готово") meta.createEl("span", { cls: "ud-card-tag", text: item.status });
  if (item.priority) meta.createEl("span", { cls: "ud-card-tag", text: item.priority });

  const due = toDate(item.due_date);
  if (due) {
    const dateBlock = card.createEl("div", { cls: "ud-date-block" });
    dateBlock.createEl("div", { cls: "ud-date-main", text: `${due.day} ${MONTHS_FULL[due.month - 1]}` });
    dateBlock.createEl("div", { cls: "ud-date-sub", text: "дедлайн" });
  }

  return {
    element: card,
    text: [item.file.name, item.type, item.status, item.priority, item.subject, item.file.folder].filter(Boolean).join(" ").toLowerCase()
  };
}

for (const [groupName, items] of groups) {
  const section = page.createEl("section", { cls: "ud-note-section" });
  const header = section.createEl("div", { cls: "ud-section" });
  header.style.setProperty("--ud-color", "var(--ud-accent-action)");
  header.createEl("span", { cls: "ud-section-dot" });
  header.createEl("span", { cls: "ud-section-title", text: groupName });
  const countEl = header.createEl("span", { cls: "ud-section-count", text: String(items.length) });

  const list = section.createEl("div", { cls: "ud-list" });
  const cards = items.map(item => renderCard(list, item));
  const emptyEl = list.createEl("div", { cls: "ud-empty", text: "Ничего не найдено." });
  emptyEl.hidden = true;

  renderedGroups.push({
    section,
    countEl,
    cards,
    defaultCount: items.length,
    emptyEl
  });
}

if (renderedGroups.length === 0) {
  page.createEl("div", { cls: "ud-empty", text: "Пока нет заметок в этом разделе." });
}

function applySearch() {
  const term = searchInput.value.trim().toLowerCase();

  for (const group of renderedGroups) {
    let visibleCount = 0;
    for (const card of group.cards) {
      const visible = !term || card.text.includes(term);
      card.element.hidden = !visible;
      if (visible) visibleCount += 1;
    }

    group.countEl.textContent = String(term ? visibleCount : group.defaultCount);
    group.emptyEl.hidden = visibleCount > 0;
    group.section.hidden = term && visibleCount === 0;
  }
}

searchInput.addEventListener("input", applySearch);
searchInput.addEventListener("keydown", event => {
  if (event.key === "Escape") {
    searchInput.value = "";
    applySearch();
    searchInput.blur();
  }
});

applySearch();
