const root = dv.container;
root.classList.add("university-dashboard");

const MONTHS_FULL = [
  "января", "февраля", "марта", "апреля", "мая", "июня",
  "июля", "августа", "сентября", "октября", "ноября", "декабря"
];

function plural(number, forms) {
  const abs = Math.abs(number) % 100;
  const last = abs % 10;
  if (abs > 10 && abs < 20) return forms[2];
  if (last > 1 && last < 5) return forms[1];
  if (last === 1) return forms[0];
  return forms[2];
}

function toDate(value) {
  if (!value) return null;
  if (typeof value === "object" && typeof value.toFormat === "function") return value;
  const parsed = DateTime.fromISO(String(value));
  return parsed.isValid ? parsed : null;
}

function openPath(path) {
  if (!path) return;
  app.workspace.openLinkText(path, "", false);
}

function makeInteractive(element, handler, label) {
  element.setAttribute("role", "button");
  element.setAttribute("tabindex", "0");
  if (label) element.setAttribute("aria-label", label);
  element.addEventListener("click", handler);
  element.addEventListener("keydown", event => {
    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      handler(event);
    }
  });
}

async function setStatus(path, status) {
  const file = app.vault.getAbstractFileByPath(path);
  if (!file) return;
  await app.fileManager.processFrontMatter(file, frontmatter => {
    frontmatter.status = status;
  });
  app.commands.executeCommandById("dataview:dataview-force-refresh-views");
}

function pillState(item, doneMode) {
  if (doneMode) {
    return {
      border: "var(--ud-accent-success)",
      pillClass: "done",
      pillText: "Готово",
      subText: item._dateLong,
      subClass: "",
      buttonText: "↩ Вернуть",
      nextStatus: "В работе"
    };
  }

  if (item._days < 0) {
    return {
      border: "var(--ud-accent-urgent)",
      pillClass: "urgent",
      pillText: "Просрочено",
      subText: `Просрочено на ${Math.abs(item._days)} дн.`,
      subClass: "ud-urgent",
      buttonText: "✓ Готово",
      nextStatus: "Готово"
    };
  }

  if (item._days === 0) {
    return {
      border: "var(--ud-status-today)",
      pillClass: "warning",
      pillText: "Сегодня",
      subText: "Сдать сегодня",
      subClass: "ud-today",
      buttonText: "✓ Готово",
      nextStatus: "Готово"
    };
  }

  if (item._days <= 3) {
    return {
      border: "var(--ud-accent-warning)",
      pillClass: "warning",
      pillText: item._days === 1 ? "Завтра" : `Через ${item._days} дн.`,
      subText: item._days === 1 ? "Остался 1 день" : `Осталось ${item._days} дн.`,
      subClass: "ud-soon",
      buttonText: "✓ Готово",
      nextStatus: "Готово"
    };
  }

  return {
    border: "var(--ud-accent-action)",
    pillClass: "info",
    pillText: "В работе",
    subText: `Через ${item._days} дн.`,
    subClass: "",
    buttonText: "✓ Готово",
    nextStatus: "Готово"
  };
}

const page = root.createEl("div", { cls: "ud-page" });
const allItems = dv.pages()
  .where(page => page.due_date)
  .array()
  .map(page => {
    const due = toDate(page.due_date);
    if (!due) return null;
    const normalized = due.startOf("day");
    const daysLeft = Math.ceil(normalized.diff(DateTime.now().startOf("day"), "days").days);
    return {
      ...page,
      _due: due,
      _days: daysLeft,
      _dateShort: `${due.day} ${MONTHS_FULL[due.month - 1]}`,
      _dateLong: `${due.day} ${MONTHS_FULL[due.month - 1]} ${due.year}`
    };
  })
  .filter(Boolean)
  .sort((a, b) => a._due.toMillis() - b._due.toMillis());

const doneItems = [];
const overdueItems = [];
const todayItems = [];
const soonItems = [];
const laterItems = [];

for (const item of allItems) {
  if (item.status === "Готово") {
    doneItems.push(item);
  } else if (item._days < 0) {
    overdueItems.push(item);
  } else if (item._days === 0) {
    todayItems.push(item);
  } else if (item._days <= 3) {
    soonItems.push(item);
  } else {
    laterItems.push(item);
  }
}

const statGrid = hero.createEl("div", { cls: "ud-stat-grid" });

const groups = [];

function registerGroup(group) {
  groups.push(group);
}

function createDeadlineCard(parent, item, doneMode = false) {
  const state = pillState(item, doneMode);
  const card = parent.createEl("div", { cls: "ud-card ud-searchable ud-focus-ring" });
  card.style.setProperty("--ud-border-color", state.border);
  makeInteractive(card, () => openPath(item.file.path), `Открыть ${item.file.name}`);

  const main = card.createEl("div", { cls: "ud-card-main" });
  main.createEl("div", { cls: "ud-card-link ud-static-title", text: item.file.name });
  const meta = main.createEl("div", { cls: "ud-card-meta" });
  if (item.subject) meta.createEl("span", { cls: "ud-card-tag", text: item.subject });
  if (item.type) meta.createEl("span", { cls: "ud-card-tag", text: item.type });

  const dateBlock = card.createEl("div", { cls: "ud-date-block" });
  dateBlock.createEl("div", { cls: "ud-date-main", text: item._dateShort });
  dateBlock.createEl("div", { cls: `ud-date-sub ${state.subClass}`.trim(), text: state.subText });

  const statusBlock = card.createEl("div", { cls: "ud-status-block" });
  statusBlock.createEl("span", { cls: `ud-task-tag ${state.pillClass}`, text: state.pillText });

  const buttonWrap = card.createEl("div", { cls: "ud-btn-block" });
  const button = buttonWrap.createEl("button", {
    cls: "ud-btn ud-focus-ring",
    text: state.buttonText,
    attr: { type: "button" }
  });
  button.addEventListener("click", async event => {
    event.stopPropagation();
    button.disabled = true;
    await setStatus(item.file.path, state.nextStatus);
  });

  return {
    element: card,
    text: [item.file.name, item.subject, item.type, item.status, item._dateLong, state.pillText, state.subText].filter(Boolean).join(" ").toLowerCase()
  };
}

function createSection(title, items, color, options = {}) {
  const section = page.createEl("section", { cls: "ud-note-section" });
  const defaultCount = items.length;
  const emptyText = options.emptyText || "Список пуст.";

  if (options.collapsible) {
    const toggle = section.createEl("div", { cls: "ud-done-toggle" });
    toggle.setAttribute("data-toggle", options.toggleKey || "done");
    toggle.createEl("span", { cls: "ud-done-arrow", text: "▶" });
    toggle.createEl("span", { cls: "ud-done-label", text: title });
    const countEl = toggle.createEl("span", { cls: "ud-done-count", text: String(defaultCount) });

    const content = section.createEl("div", { cls: "ud-done-list" });
    toggle.addEventListener("click", () => {
      toggle.classList.toggle("open");
      content.classList.toggle("open");
    });

    const list = content.createEl("div", { cls: "ud-list" });
    const cards = items.map(item => createDeadlineCard(list, item, true));
    const emptyEl = list.createEl("div", { cls: "ud-empty", text: emptyText });
    emptyEl.hidden = cards.length > 0;

    registerGroup({ section, countEl, cards, emptyEl, defaultCount, emptyText, collapsible: true, content, toggle });
    return;
  }

  const header = section.createEl("div", { cls: "ud-section" });
  header.style.setProperty("--ud-color", color);
  header.createEl("span", { cls: "ud-section-dot" });
  header.createEl("span", { cls: "ud-section-title", text: title });
  const countEl = header.createEl("span", { cls: "ud-section-count", text: String(defaultCount) });

  const list = section.createEl("div", { cls: "ud-list" });
  const cards = items.map(item => createDeadlineCard(list, item, false));
  const emptyEl = list.createEl("div", { cls: "ud-empty", text: emptyText });
  emptyEl.hidden = cards.length > 0;

  registerGroup({ section, countEl, cards, emptyEl, defaultCount, emptyText, collapsible: false });
}

function applySearch() {
  const term = searchInput.value.trim().toLowerCase();

  for (const group of groups) {
    let visibleCount = 0;

    for (const card of group.cards) {
      const visible = !term || card.text.includes(term);
      card.element.hidden = !visible;
      if (visible) visibleCount += 1;
    }

    group.countEl.textContent = String(term ? visibleCount : group.defaultCount);

    if (group.defaultCount === 0) {
      group.emptyEl.hidden = false;
      group.emptyEl.textContent = group.emptyText;
      group.section.hidden = false;
      continue;
    }

    group.emptyEl.hidden = visibleCount > 0;
    group.emptyEl.textContent = term ? "Ничего не найдено." : group.emptyText;
    group.section.hidden = term && visibleCount === 0;

    if (group.collapsible) {
      if (term && visibleCount > 0) {
        group.content.classList.add("open");
        group.toggle.classList.add("open");
      } else if (!term) {
        group.content.classList.remove("open");
        group.toggle.classList.remove("open");
      }
    }
  }
}

searchInput.addEventListener("input", applySearch);
searchInput.addEventListener("keydown", event => {
  if (event.key === "Escape") {
    searchInput.value = "";
    applySearch();
    searchInput.blur();
  }
});

applySearch();
