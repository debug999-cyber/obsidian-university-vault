# Дедлайны

```dataviewjs
/* =============================================================================
   DEADLINE DASHBOARD — DataviewJS
   -----------------------------------------------------------------------------
   Назначение:  Автоматическая панель дедлайнов. Сканирует всё хранилище,
                находит заметки с полем due_date, группирует по срочности
                и позволяет менять статус прямо из таблицы.
   
   Требования:  Obsidian + плагин Dataview + CSS-классы .university-dashboard
                и цветовые переменные (см. Блок 4).
   
   Как вставить: Создай заметку → вставь этот блок в ```dataviewjs ... ```
   
   Версия:      2.0 (2026-05-24)
   
   Архитектура:
     Блок 1 — Конфигурация       (константы, пороги, переводы месяцев)
     Блок 2 — Сбор данных        (Dataview-запрос и классификация)
     Блок 3 — Утилиты            (смена статуса, принудительный рефреш)
     Блок 4 — Компонент карточки (рендеринг одной задачи с цветовой логикой)
     Блок 5 — Компоненты секций  (секция + аккордеон «Выполнено»)
     Блок 6 — Монтаж             (вызовы рендеринга в нужном порядке)
   ============================================================================= */

// ============================================
// БЛОК 1: КОНФИГУРАЦИЯ И КОНСТАНТЫ
// ============================================

/** Текущая дата, обнулённая до 00:00. Используется как точка отсчёта
 *  для всех вычислений «осталось дней». startOf("day") нужен, чтобы
 *  сравнение было чисто календарным (не учитывать текущее время). */
const сегодня = DateTime.now().startOf("day");

/** Родительные падежи месяцев для красивого вывода: «24 мая 2026».
 *  Индекс 0 → январь. Если нужен именительный падеж, замените массив. */
const МЕСЯЦЫ = [
    "января", "февраля", "марта", "апреля", "мая", "июня",
    "июля", "августа", "сентября", "октября", "ноября", "декабря"
];

/** Порог «скоро» в днях. Задачи с дедлайном через 1–2 дня попадут
 *  в секцию «Скоро дедлайн» с оранжевым акцентом.
 *  Измените число, если хотите другой диапазон. */
const ПОРОГ_СКОРО = 2;


// ============================================
// БЛОК 2: СБОР ДАННЫХ И КЛАССИФИКАЦИЯ
// ============================================

/** Получаем все страницы хранилища, у которых в frontmatter есть due_date.
 *  Затем распределяем по 6 корзинам:
 *   • выполнено   — status === "Готово" (не важно, просрочено ли)
 *   • просрочено  — активные с дедлайном в прошлом
 *   • сегодняЗам  — активные, дедлайн сегодня
 *   • скоро       — активные, дедлайн через 1–ПОРОГ_СКОРО дней
 *   • ок          — активные, дедлайн позже ПОРОГ_СКОРО
 *   • активные    — объединение просрочено + сегодня + скоро + ок
 *                   (нужен для статистики, хотя сейчас не выводится) */

const всеЗаметки = dv.pages().where(p => p.due_date).array();
const активные   = [];
const выполнено  = [];
const просрочено = [];
const сегодняЗам = [];
const скоро      = [];
const ок         = [];

for (const p of всеЗаметки) {
    // Парсим ISO-дату из frontmatter и вычисляем разницу в днях
    const д = DateTime.fromISO(p.due_date);
    const дней = Math.ceil(д.diff(сегодня, "days").days);

    // Вешаем вычисленные поля на объект страницы, чтобы не считать
    // повторно в функции карточки. _дней и _дата — «приватные» поля.
    p._дней = дней;
    p._дата = `${д.day} ${МЕСЯЦЫ[д.month - 1]} ${д.year}`;

    // Разветвление: выполненные идут отдельно, остальные — по срочности
    if (p.status === "Готово") {
        выполнено.push(p);
        continue;
    }

    активные.push(p);

    if (дней < 0)        просрочено.push(p);
    else if (дней === 0) сегодняЗам.push(p);
    else if (дней <= ПОРОГ_СКОРО) скоро.push(p);
    else                 ок.push(p);
}

/** Сортировка по дате дедлайна (от ближайшего к дальнему внутри секции).
 *  Выполненные тоже сортируются, чтобы в аккордеоне был порядок. */
const сортПоДате = (a, b) => a.due_date < b.due_date ? -1 : 1;
[выполнено, просрочено, сегодняЗам, скоро, ок].forEach(arr => arr.sort(сортПоДате));


// ============================================
// БЛОК 3: УТИЛИТЫ (API Obsidian)
// ============================================

/** Контейнер Dataview — сюда будет монтироваться весь DOM.
 *  Класс .university-dashboard нужен для применения кастомных стилей. */
const container = dv.container;
container.classList.add("university-dashboard");

/**
 * Меняет поле status в frontmatter файла и принудительно обновляет
 * все Dataview-блоки в рабочей области.
 *
 * Важно: processFrontMatter работает асинхронно и требует абстрактный
 * файл (TFile), а не просто путь. Если файл не найден — молча выходит.
 *
 * @param {string} путь   — путь в хранилище, например "10-Университет/Биология/..."
 * @param {string} новыйСтатус — значение для поля status ("Готово", "В работе" и т.д.)
 */
async function сменитьСтатус(путь, новыйСтатус) {
    const файл = app.vault.getAbstractFileByPath(путь);
    if (!файл) return;

    await app.fileManager.processFrontMatter(файл, (fm) => {
        fm.status = новыйСтатус;
    });

    // Принудительный рефреш всех Dataview-блоков, чтобы изменение
    // статуса отобразилось мгновенно без ручного обновления страницы.
    app.commands.executeCommandById("dataview:dataview-force-refresh-views");
}


// ============================================
// БЛОК 4: КОМПОНЕНТ «КАРТОЧКА»
// ============================================
/**
 * Рендерит одну задачу в виде горизонтальной карточки.
 *
 * Цветовая логика (CSS-переменные должны быть определены в теме):
 *   --ud-status-overdue  → красный  (просрочено)
 *   --ud-status-today    → жёлтый/оранжевый (сегодня)
 *   --ud-status-soon     → оранжевый (1–2 дня)
 *   --ud-status-ok       → зелёный/серый (всё остальное)
 *   --ud-status-done     → приглушённый (выполнено)
 *
 * Интерактивность:
 *   • Двойной клик по карточке (не по кнопке) → открыть заметку
 *   • Клик по названию → открыть заметку
 *   • Кнопка «Готово» / «↩ Вернуть» → переключает статус через Блок 3
 *
 * @param {object} p        — объект страницы Dataview с полями _дней, _дата, file, subject, type, status...
 * @param {HTMLElement} список — родительский DOM-элемент (div), куда вставлять карточку
 * @param {boolean} showDone  — если true, карточка рисуется в режиме «выполнено»
 *                            (приглушённые цвета, кнопка «Вернуть»)
 */
function карточка(p, список, showDone = false) {
    const дней = p._дней;

    // --- Вычисляем визуальные атрибуты на основе сроков ---
    let borderColor, pillClass, subText, subClass;

    if (showDone) {
        // Выполненные задачи: нейтральный стиль
        borderColor = "var(--ud-status-done)";
        pillClass   = "ud-pill-done";
        subText     = p._дата;
        subClass    = "";
    } else if (дней < 0) {
        // Просрочено
        borderColor = "var(--ud-status-overdue)";
        pillClass   = "ud-pill-overdue";
        subText     = `Просрочено на ${Math.abs(дней)} дн.`;
        subClass    = "ud-urgent";
    } else if (дней === 0) {
        // Дедлайн сегодня
        borderColor = "var(--ud-status-today)";
        pillClass   = "ud-pill-today";
        subText     = "Сегодня";
        subClass    = "ud-today";
    } else if (дней <= ПОРОГ_СКОРО) {
        // Дедлайн скоро (1–2 дня)
        borderColor = "var(--ud-status-soon)";
        pillClass   = "ud-pill-soon";
        subText     = дней === 1 ? "Завтра" : `Через ${дней} дн.`;
        subClass    = "ud-soon";
    } else {
        // Всё под контролем
        borderColor = "var(--ud-status-ok)";
        pillClass   = "ud-pill-ok";
        subText     = `Через ${дней} дн.`;
        subClass    = "";
    }

    // --- DOM: карточка ---
    const row = список.createEl("div", { cls: "ud-card" });
    row.style.setProperty("--ud-border-color", borderColor);
    row.title = "Двойной клик — открыть задачу";

    // Двойной клик по карточке открывает заметку, но не если кликнули
    // по кнопке (чтобы случайно не открыть файл при нажатии «Готово»)
    row.addEventListener("dblclick", (e) => {
        if (e.target.closest("button")) return;
        app.workspace.openLinkText(p.file.path, "", false);
    });

    // --- Левая часть: название и метки ---
    const main = row.createEl("div", { cls: "ud-card-main" });

    const link = main.createEl("a", { text: p.file.name, cls: "ud-card-link" });
    link.addEventListener("click", (e) => {
        e.stopPropagation();
        app.workspace.openLinkText(p.file.path, "", false);
    });

    const meta = main.createEl("div", { cls: "ud-card-meta" });
    [p.subject, p.type].forEach(текст => {
        if (!текст) return;
        meta.createEl("span", { text: текст, cls: "ud-card-tag" });
    });

    // --- Правая часть: дата и срок ---
    const dateBlock = row.createEl("div", { cls: "ud-date-block" });
    dateBlock.createEl("div", { text: p._дата, cls: "ud-date-main" });
    dateBlock.createEl("div", { text: subText, cls: `ud-date-sub ${subClass}` });

    // --- Статус-пилюля ---
    const statusBlock = row.createEl("div", { cls: "ud-status-block" });
    const pillText = showDone
        ? "Готово"
        : (дней < 0 ? "Просрочено" : дней === 0 ? "В срок" : дней <= ПОРОГ_СКОРО ? "Скоро" : "В работе");

    statusBlock.createEl("span", {
        text: pillText,
        cls: `ud-pill ${pillClass}`
    });

    // --- Кнопка переключения статуса ---
    const btnBlock = row.createEl("div", { cls: "ud-btn-block" });
    const btn = btnBlock.createEl("button", {
        text: showDone ? "↩ Вернуть" : "✓ Готово",
        cls: "ud-btn"
    });

    btn.addEventListener("click", async (e) => {
        e.stopPropagation();
        // Переключаем статус: Готово ↔ В работе
        const новыйСтатус = showDone ? "В работе" : "Готово";
        await сменитьСтатус(p.file.path, новыйСтатус);
    });
}


// ============================================
// БЛОК 5: КОМПОНЕНТЫ СЕКЦИЙ
// ============================================

/**
 * Рендерит секцию с заголовком и списком карточек.
 * Если список пуст — ничего не создаёт (секция не отображается).
 *
 * @param {Array}  список   — массив объектов страниц Dataview
 * @param {string} название — заголовок секции («Просрочено», «Сегодня»...)
 * @param {string} цветVar  — CSS-переменная для цвета точки и рамки
 * @param {boolean} showDone — передаётся в карточку (см. Блок 4)
 */
function секция(список, название, цветVar, showDone = false) {
    if (список.length === 0) return;

    // Заголовок секции с цветной точкой и счётчиком
    const header = container.createEl("div", { cls: "ud-section" });
    header.style.setProperty("--ud-color", `var(${цветVar})`);
    header.createEl("span", { cls: "ud-section-dot" });
    header.createEl("span", { text: название, cls: "ud-section-title" });
    header.createEl("span", { text: String(список.length), cls: "ud-section-count" });

    // Контейнер карточек
    const list = container.createEl("div", { cls: "ud-list" });
    список.forEach(p => карточка(p, list, showDone));
}

/**
 * Рендерит сворачиваемую секцию «Выполнено».
 * По умолчанию скрыта (аккордеон). Клик по заголовку переключает видимость.
 *
 * Реализация: используем CSS-классы .open на контейнере и заголовке.
 * Стили должны содержать правило .ud-done-list:not(.open) { display: none; }
 *
 * @param {Array} список — массив выполненных задач
 */
function секцияВыполнено(список) {
    if (список.length === 0) return;

    const wrap = container.createEl("div");

    // Заголовок-переключатель
    const toggle = wrap.createEl("div", { cls: "ud-done-toggle" });
    toggle.createEl("span", { text: "▶", cls: "ud-done-arrow" });
    toggle.createEl("span", { text: "Выполнено", cls: "ud-done-label" });
    toggle.createEl("span", { text: String(список.length), cls: "ud-done-count" });

    // Содержимое, изначально скрыто через CSS
    const content = wrap.createEl("div", { cls: "ud-done-list" });

    toggle.addEventListener("click", () => {
        content.classList.toggle("open");
        toggle.classList.toggle("open");
    });

    список.forEach(p => карточка(p, content, true));
}


// ============================================
// БЛОК 6: МОНТАЖ (рендеринг в заданном порядке)
// ============================================
// Порядок секций важен для восприятия: от самого критичного
// к нейтральному, затем свёрнутая «Выполнено».

секция(просрочено, "Просрочено",      "--ud-status-overdue");
секция(сегодняЗам, "Сегодня",         "--ud-status-today");
секция(скоро,      "Скоро дедлайн",   "--ud-status-soon");
секция(ок,         "Под контролем",   "--ud-status-ok");
секцияВыполнено(выполнено);

// Плейсхолдер, если в хранилище вообще нет заметок с due_date
if (всеЗаметки.length === 0) {
    container.createEl("div", {
        text: "Нет заметок с дедлайнами",
        cls: "ud-empty"
    });
}
