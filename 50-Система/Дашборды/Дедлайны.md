# Дедлайны

```dataviewjs
const root = dv.container;
root.classList.add("university-dashboard", "ud-page");
root.innerHTML = "";

const today = DateTime.now().startOf("day");
const months = ["янв", "фев", "мар", "апр", "мая", "июн", "июл", "авг", "сен", "окт", "ноя", "дек"];
const soonLimit = 2;
function toDate(value) {
  if (!value) return null;
  if (value.toISODate) return value.startOf ? value.startOf("day") : DateTime.fromISO(value.toISODate()).startOf("day");
  const d = DateTime.fromISO(String(value).replaceAll('"', '').trim());
  return d.isValid ? d.startOf("day") : null;
}
function dateText(d) { return d ? `${d.day} ${months[d.month - 1]} ${d.year}` : "без даты"; }
function daysLeft(d) { return Math.ceil(d.diff(today, "days").days); }
function dayLabel(days) {
  if (days < 0) return `Просрочено на ${Math.abs(days)} дн.`;
  if (days === 0) return "Сегодня";
  if (days === 1) return "Завтра";
  return `Через ${days} дн.`;
}
function color(days, done=false) {
  if (done) return "--ud-status-done";
  if (days < 0) return "--ud-status-overdue";
  if (days === 0) return "--ud-status-today";
  if (days <= soonLimit) return "--ud-status-soon";
  return "--ud-status-ok";
}
function openPath(path) { app.workspace.openLinkText(path, "", false); }
async function setStatus(path, status) {
  const file = app.vault.getAbstractFileByPath(path);
  if (!file) return;
  await app.fileManager.processFrontMatter(file, fm => { fm.status = status; });
  app.commands.executeCommandById("dataview:dataview-force-refresh-views");
}
function renderSection(title, items, colorVar, initiallyOpen=true) {
  const wrap = root.createEl("div", { cls: "ud-block" });
  const toggle = wrap.createEl("div", { cls: `ud-done-toggle ${initiallyOpen ? "open" : ""}` });
  toggle.style.setProperty("--ud-color", `var(${colorVar})`);
  toggle.createEl("span", { text: "▶", cls: "ud-done-arrow" });
  toggle.createEl("span", { text: title, cls: "ud-done-label" });
  toggle.createEl("span", { text: String(items.length), cls: "ud-done-count" });
  const list = wrap.createEl("div", { cls: `ud-done-list ${initiallyOpen ? "open" : ""}` });
  toggle.addEventListener("click", () => { toggle.classList.toggle("open"); list.classList.toggle("open"); });
  if (!items.length) list.createEl("div", { text: "Пусто", cls: "ud-empty" });
  for (const p of items) renderCard(list, p, p.status === "Готово");
}
function renderCard(parent, p, done=false) {
  const days = p._days;
  const row = parent.createEl("div", { cls: "ud-card" });
  row.style.setProperty("--ud-border-color", `var(${color(days, done)})`);
  row.dataset.search = [p.file.name, p.subject, p.type, p.status, p.file.folder].filter(Boolean).join(" ").toLowerCase();
  row.addEventListener("dblclick", e => { if (!e.target.closest("button")) openPath(p.file.path); });
  const main = row.createEl("div", { cls: "ud-card-main" });
  const link = main.createEl("a", { text: p.file.name, cls: "ud-card-link" });
  link.addEventListener("click", e => { e.stopPropagation(); openPath(p.file.path); });
  const meta = main.createEl("div", { cls: "ud-card-meta" });
  for (const t of [p.subject, p.type, p.priority, p.status]) if (t) meta.createEl("span", { text: String(t), cls: "ud-card-tag" });
  const date = row.createEl("div", { cls: "ud-date-block" });
  date.createEl("div", { text: dateText(p._date), cls: "ud-date-main" });
  date.createEl("div", { text: done ? "выполнено" : dayLabel(days), cls: `ud-date-sub ${days < 0 ? "ud-urgent" : days === 0 ? "ud-today" : days <= soonLimit ? "ud-soon" : ""}` });
  const st = row.createEl("div", { cls: "ud-status-block" });
  const btn = st.createEl("button", { text: done ? "Вернуть" : "Готово", cls: `ud-pill ${done ? "ud-pill-done" : "ud-pill-ok"}` });
  btn.addEventListener("click", e => { e.stopPropagation(); setStatus(p.file.path, done ? "В работе" : "Готово"); });
}

const pages = dv.pages().where(p => p.due_date).array()
  .map(p => ({ ...p, _date: toDate(p.due_date) }))
  .filter(p => p._date)
  .map(p => ({ ...p, _days: daysLeft(p._date) }))
  .sort((a,b) => a._date - b._date);
const active = pages.filter(p => p.status !== "Готово");
const done = pages.filter(p => p.status === "Готово").sort((a,b) => b._date - a._date);
const groups = {
  overdue: active.filter(p => p._days < 0),
  today: active.filter(p => p._days === 0),
  soon: active.filter(p => p._days > 0 && p._days <= soonLimit),
  ok: active.filter(p => p._days > soonLimit)
};

const hero = root.createEl("div", { cls: "ud-hero" });
const left = hero.createEl("div");
left.createEl("h1", { text: "Дедлайны" });
left.createEl("p", { text: "Автоматический контроль сроков по полю due_date. Кнопка в карточке меняет status в frontmatter." });
const stats = hero.createEl("div", { cls: "ud-stat-grid" });
for (const s of [["Активные", active.length, "--ud-blue"], ["Просрочено", groups.overdue.length, "--ud-status-overdue"], ["Сегодня", groups.today.length, "--ud-status-today"], ["Выполнено", done.length, "--ud-status-done"]]) {
  const item = stats.createEl("div", { cls: "ud-stat" });
  item.style.setProperty("--ud-color", `var(${s[2]})`);
  item.createEl("strong", { text: String(s[1]) });
  item.createEl("span", { text: s[0] });
}
const search = root.createEl("div", { cls: "ud-search" });
search.createEl("span", { text: "Поиск", cls: "ud-search-label" });
const input = search.createEl("input", { type: "search", placeholder: "Фильтр по дедлайнам..." });
input.addEventListener("input", () => {
  const term = input.value.trim().toLowerCase();
  root.querySelectorAll(".ud-card[data-search]").forEach(el => el.style.display = !term || el.dataset.search.includes(term) ? "" : "none");
});

renderSection("Просрочено", groups.overdue, "--ud-status-overdue", true);
renderSection("Сегодня", groups.today, "--ud-status-today", true);
renderSection("Скоро", groups.soon, "--ud-status-soon", true);
renderSection("Под контролем", groups.ok, "--ud-status-ok", true);
renderSection("Выполнено", done.slice(0, 30), "--ud-status-done", false);
```
