   -----------------------------------------------------------------------------
   Назначение:  Создаёт структурированную заметку для университета/проектов
                с автоматической генерацией MOC (Map of Content) и связями.
   
   Требования:  Obsidian + плагины:

// ============================================

/**
 * Проверяет строку даты на соответствие формату ГГГГ-ММ-ДД.
 * Важно: new Date("2024-02-30") создаёт 1 марта, поэтому делаем
 * строгую проверку компонентов (год/месяц/день) после парсинга.
 */
function isValidDate(dateString) {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(dateString)) return false;
    const date = new Date(dateString);
    if (isNaN(date.getTime())) return false;
    const [y, m, d] = dateString.split('-').map(Number);
    return date.getFullYear() === y && (date.getMonth() + 1) === m && date.getDate() === d;
}


// ============================================
// БЛОК 2: КОНФИГУРАЦИЯ
// ============================================
// Централизованные справочники. При добавлении нового предмета или типа
// достаточно изменить только эти массивы — остальная логика подхватит
// автоматически.

/** Список предметов/областей. Специальные значения:
 *  "IT"      → попадает в папку 20-Проекты/IT
 *  "Личное"  → попадает в папку 30-Личное
 *  Остальные → попадают в 10-Университет/<предмет>
 */
const ПРЕДМЕТЫ = [
    "ВВС", "ОП", "Физика", "Информатика", "Математика",
    "Английский", "Физра", "ПИД", "Биология", "Русский",
    "УНПАО", "Литература", "История", "Обществознание",
    "География", "IT", "Личное", "Другое"
];

/** Типы заметок. Используются для:
 *  • выбора подпапки (Блок 4)
 *  • определения нужен ли дедлайн (см. ТИПЫ_С_ДЕДЛАЙНОМ)
 *  • раскраски карточек в MOC (Блок 6)
 */
const ТИПЫ = [
    "Проект", "ДЗ", "Лекция", "Конспект",
    "Лаба", "Экзамен", "Источник", "Разное"
];

/** Типы, для которых Templater запросит дедлайн, статус и приоритет.
 *  Остальные типы автоматически получают статус "Готово".
 */
const ТИПЫ_С_ДЕДЛАЙНОМ = ["Проект", "ДЗ", "Лаба", "Экзамен"];


// ============================================
// БЛОК 3: СБОР ДАННЫХ ОТ ПОЛЬЗОВАТЕЛЯ
// ============================================
// Последовательные диалоги. Если пользователь нажмёт Esc на любом этапе,
// скрипт прерывается через `return`, и заметка не создаётся.

const предмет = await tp.system.suggester(ПРЕДМЕТЫ, ПРЕДМЕТЫ, false, "Выбери предмет:");
if (!предмет) return;  // Отмена

const тип = await tp.system.suggester(ТИПЫ, ТИПЫ, false, "Выбери тип:");
if (!тип) return;      // Отмена

const тема = await tp.system.prompt("Введи тему:", "");
if (!тема) return;     // Отмена (пустая тема не допускается)

// Определяем, нужно ли собирать мета-данные задачи (дедлайн/статус/приоритет)
const нуженДедлайн = ТИПЫ_С_ДЕДЛАЙНОМ.includes(тип);

let дедлайн   = "";
let статус    = "";
let приоритет = "";

if (нуженДедлайн) {
    // --- Дедлайн с валидацией ---
    // Цикл повторяется, пока пользователь не введёт корректную дату
    // или не отменит диалог (пустой ввод → return).
    let promptText = "Дедлайн (ГГГГ-ММ-ДД):";
    while (true) {
        дедлайн = await tp.system.prompt(promptText, tp.date.now("YYYY-MM-DD"));
        if (!дедлайн) return;          // Отмена
        if (isValidDate(дедлайн)) break; // Успех — выходим из цикла
        promptText = "Неверная дата! Дедлайн (ГГГГ-ММ-ДД):";
    }

    // --- Статус ---
    const СТАТУСЫ = ["Идея", "В работе", "На проверке", "Готово"];
    статус = await tp.system.suggester(СТАТУСЫ, СТАТУСЫ, false, "Выбери статус:");
    if (!статус) return;

    // --- Приоритет ---
    // Значения low/medium/high используются в Dataview-фильтрах и CSS.
    const ПРИОРИТЕТЫ_ТЕКСТ = ["Высокий", "Средний", "Низкий"];
    const ПРИОРИТЕТЫ_ЗНАЧ  = ["high", "medium", "low"];
    приоритет = await tp.system.suggester(ПРИОРИТЕТЫ_ТЕКСТ, ПРИОРИТЕТЫ_ЗНАЧ, false, "Приоритет:");
    if (!приоритет) приоритет = "medium";  // Значение по умолчанию
} else {
    // Для лекций/конспектов/источников статус всегда "Готово"
    статус = "Готово";
}


// ============================================
// БЛОК 4: ФОРМИРОВАНИЕ ПУТЕЙ
// ============================================
// Здесь определяется, в какие папки попадут файл и MOC.
// 
// Правила маппинга базового пути:
//   IT      → 20-Проекты/IT
//   Личное  → 30-Личное
//   Иное    → 10-Университет/<Предмет>
//
// Правила маппинга подпапки по типу:
//   Проект   → Проекты/Проект <Тема>  (уникальная папка под проект)
//   ДЗ       → ДЗ
//   Лекция   → Лекции
//   Конспект → Конспекты
//   Лаба     → Лабораторные
//   Экзамен  → Экзамены
//   Источник → Источники
//   Разное   → Разное

let базовыйПуть = "";
if (предмет === "IT") {
    базовыйПуть = "20-Проекты/IT";
} else if (предмет === "Личное") {
    базовыйПуть = "30-Личное";
} else {
    базовыйПуть = "10-Университет/" + предмет;
}

let папкаТипа = "";
if (тип === "Проект") {
    папкаТипа = "Проекты/Проект " + тема;
} else if (тип === "ДЗ") {
    папкаТипа = "ДЗ";
} else if (тип === "Лекция") {
    папкаТипа = "Лекции";
} else if (тип === "Конспект") {
    папкаТипа = "Конспекты";
} else if (тип === "Лаба") {
    папкаТипа = "Лабораторные";
} else if (тип === "Экзамен") {
    папкаТипа = "Экзамены";
} else if (тип === "Источник") {
    папкаТипа = "Источники";
} else {
    папкаТипа = "Разное";
}

// Полный путь до целевой папки (без имени файла)
const итоговыйПуть = базовыйПуть + "/" + папкаТипа;


// ============================================
// БЛОК 5: СОЗДАНИЕ ПАПОК
// ============================================
// Obsidian НЕ создаёт промежуточные папки автоматически при записи файла.
// Поэтому создаём их явно через app.vault.createFolder().
// 
// Обертка try/catch подавляет ошибку "папка уже существует",
// чтобы не прерывать выполнение, если структура уже есть.

try {
    await app.vault.createFolder(базовыйПуть);
} catch (e) {
    // Игнорируем: папка уже существует или путь некорректен
}

try {
    await app.vault.createFolder(итоговыйПуть);
} catch (e) {
    // Игнорируем: папка уже существует
}


// ============================================
// БЛОК 6: СОЗДАНИЕ MOC (Map of Content)
// ============================================
// MOC — это дашборд-предмет, который автоматически собирает все заметки
// предмета через DataviewJS и группирует их по типам с цветовой индикацией.
//
// ВАЖНЫЕ ТЕХНИЧЕСКИЕ ДЕТАЛИ:
// 1. Мы НЕ передаём путь в dv.pages("строка"), потому что Dataview
//    имеет собственный query-parser, который ломается на путях с цифрами
//    и кириллицей (ошибка: "Expected string, tag or file link").
//    Вместо этого используем dv.pages() без аргументов + .where() с JS-фильтром.
//
// 2. Для вставки строковых литералов внутрь генерируемого JS-кода
//    используем JSON.stringify() — это гарантирует правильное экранирование
//    кавычек и спецсимволов. Раньше использовался String.fromCharCode(34),
//    что было хрупким решением.
//
// 3. Переменная `cb` хранит "```", чтобы не конфликтовать с парсером
//    Templater/JS (раньше использовалось имя codeBlock, которое иногда
//    интерпретировалось некорректно).

const mocPath = базовыйПуть + "/MOC — " + предмет + ".md";
const mocExists = app.vault.getAbstractFileByPath(mocPath);

if (!mocExists) {
    // JSON.stringify() оборачивает путь в кавычки и экранирует содержимое.
    // Пример: "10-Университет/Биология" → '"10-Университет/Биология"'
    const basePathLit = JSON.stringify(базовыйПуть);
    const selfNameLit = JSON.stringify("MOC — " + предмет);
    const cb = "```";  // Маркер начала/конца code-block

    // Строим содержимое MOC построчно через массив.
    // .join("\n") гарантирует корректные переносы строк без лишних \r.
    const mocContent = [
        "---",
        "cssclasses:",
        "  - university-dashboard",
        "---",
        "",
        "> Map of Content для предмета «" + предмет + "». Автоматически собирает все заметки.",
        "",
        cb + "dataviewjs",
        "const container = dv.container;",
        "container.classList.add(\"university-dashboard\");",
        "",
        // Сбор страниц: берём ВСЕ заметки хранилища, затем фильтруем
        // по началу пути и исключаем сам MOC, чтобы избежать рекурсии.
        "const pages = dv.pages()",
        "    .where(p => p.file.path.startsWith(" + basePathLit + "))",
        "    .where(p => p.file.name != " + selfNameLit + ")",
        "    .sort(p => p.file.name)",
        "    .array();",
        "",
        // CSS-переменные для раскраски карточек по типу.
        // Добавить новый цвет: внеси пару в этот объект и определи
        // CSS-переменную --ud-<цвет> в своей теме.
        "const typeColors = {",
        '    "ДЗ": "--ud-blue",',
        '    "Лекция": "--ud-green",',
        '    "Конспект": "--ud-teal",',
        '    "Проект": "--ud-orange",',
        '    "Лаба": "--ud-purple",',
        '    "Экзамен": "--ud-red",',
        '    "Источник": "--ud-frost",',
        '    "Разное": "--ud-frost"',
        "};",
        "",
        // Группировка страниц по типу (p.type из frontmatter)
        "const byType = {};",
        "for (const p of pages) {",
        '    const t = p.type || "Разное";',
        "    if (!byType[t]) byType[t] = [];",
        "    byType[t].push(p);",
        "}",
        "",
        // Вспомогательные переменные для отображения дедлайнов
        "const сегодня = DateTime.now().startOf(\"day\");",
        'const месяцы = ["янв","фев","мар","апр","мая","июн","июл","авг","сен","окт","ноя","дек"];',
        "",
        // Рендеринг секций по типам
        "for (const [type, items] of Object.entries(byType)) {",
        '    const colorVar = typeColors[type] || "--ud-frost";',
        '    const section = container.createEl("div", { cls: "ud-moc-section" });',
        '    const header = section.createEl("div", { cls: "ud-moc-header ud-section" });',
        '    header.style.setProperty("--ud-color", "var(" + colorVar + ")");',
        '    header.createEl("span", { cls: "ud-section-dot" });',
        '    const title = header.createEl("span", { text: type, cls: "ud-section-title" });',
        '    title.style.textTransform = "uppercase";',
        '    header.createEl("span", { text: String(items.length), cls: "ud-section-count" });',
        '    const list = section.createEl("div", { cls: "ud-moc-list ud-list" });',
        "",
        // Рендеринг карточки отдельной заметки
        "    for (const p of items) {",
        '        const card = list.createEl("div", { cls: "ud-card" });',
        '        card.style.setProperty("--ud-border-color", "var(" + colorVar + ")");',
        "        card.addEventListener(\"dblclick\", () => app.workspace.openLinkText(p.file.path, \"\", false));",
        "",
        '        const main = card.createEl("div", { cls: "ud-card-main" });',
        '        const link = main.createEl("a", { text: p.file.name, cls: "ud-card-link" });',
        "        link.addEventListener(\"click\", (e) => {",
        "            e.stopPropagation();",
        "            app.workspace.openLinkText(p.file.path, \"\", false);",
        "        });",
        "",
        '        const meta = main.createEl("div", { cls: "ud-card-meta" });',
        "",
        // Статус (если не "Готово")
        "        if (p.status && p.status !== \"Готово\") {",
        '            const statusTag = meta.createEl("span", { text: p.status, cls: "ud-card-tag" });',
        '            if (p.status === "В работе") statusTag.style.color = "var(--ud-status-work)";',
        '            if (p.status === "На проверке") statusTag.style.color = "var(--ud-status-today)";',
        '            if (p.status === "Идея") statusTag.style.color = "var(--ud-status-soon)";',
        "        }",
        "",
        // Дедлайн с цветовой индикацией срочности
        "        if (p.due_date) {",
        "            const д = DateTime.fromISO(p.due_date);",
        '            const дней = Math.ceil(д.diff(сегодня, "days").days);',
        '            const dateText = д.day + " " + месяцы[д.month-1];',
        '            const dateTag = meta.createEl("span", { text: dateText, cls: "ud-card-tag" });',
        '            if (дней < 0) dateTag.style.color = "var(--ud-status-overdue)";',
        '            else if (дней === 0) dateTag.style.color = "var(--ud-status-today)";',
        '            else if (дней <= 2) dateTag.style.color = "var(--ud-status-soon)";',
        "        }",
        "    }",
        "}",
        "",
        // Плейсхолдер при отсутствии заметок
        "if (pages.length === 0) {",
        '    container.createEl("div", { text: "Пока нет заметок", cls: "ud-empty" });',
        "}",
        cb,  // Закрывающий ```
        ""
    ].join("\n");

    // Создаём MOC с обработкой ошибок. Если что-то пойдёт не так
    // (например, нет прав на запись), пользователь увидит Notice.
    try {
        await app.vault.create(mocPath, mocContent);
    } catch (e) {
        console.error("MOC creation failed:", e);
        new Notice("Не удалось создать MOC: " + e.message);
    }
}


// ============================================
// БЛОК 7: ПОИСК ПРЕДЫДУЩЕЙ ЗАМЕТКИ (связи)
// ============================================
// Для лекций и конспектов автоматически находим последнюю созданную
// заметку этого же предмета и вставляем ссылку на неё.
// 
// Фильтры исключают:
//   • служебные файлы (начинаются с "_")
//   • сам MOC (начинается с "MOC")
//   • индекс-файлы (начинаются с "Index")
//
// Сортировка по имени в обратном порядке (localeCompare) даёт
// хронологический порядок, если дата стоит в начале имени файла.

const имяФайла = tp.date.now("YYYY-MM-DD") + " — " + тема;

let предыдущаяСсылка = "";

if (["Лекция", "Конспект"].includes(тип)) {
    const всеФайлы = app.vault.getMarkdownFiles();
    const файлыПредмета = всеФайлы
        .filter(f => f.path.startsWith(базовыйПуть + "/")
            && !f.name.startsWith("_")
            && !f.name.startsWith("MOC")
            && !f.name.startsWith("Index"))
        .sort((a, b) => b.name.localeCompare(a.name));

    if (файлыПредмета.length > 0) {
        // Формат вики-ссылки Obsidian: [[Имя файла без расширения]]
        предыдущаяСсылка = "[[" + файлыПредмета[0].basename + "]]";
    }
}


// ============================================
// БЛОК 8: ФОРМИРОВАНИЕ ТЕГОВ И YAML-ПОЛЕЙ
// ============================================
// Теги строятся иерархически для удобной фильтрации в Dataview:
//   университет/биология/конспект
//   университет/биология
//   университет
//   биология
//   конспект
//
// Для IT и Личного используются отдельные пространства имён.

let теги = [];

if (предмет === "IT") {
    теги = [
        "проекты/it/" + тип.toLowerCase(),
        "проекты/it",
        тип.toLowerCase()
    ];
} else if (предмет === "Личное") {
    теги = [
        "личное/" + тип.toLowerCase(),
        "личное",
        тип.toLowerCase()
    ];
} else {
    теги = [
        "университет/" + предмет.toLowerCase() + "/" + тип.toLowerCase(),
        "университет/" + предмет.toLowerCase(),
        "университет",
        предмет.toLowerCase(),
        тип.toLowerCase()
    ];
}

// Формируем YAML-списки и опциональные поля.
// Пустые поля (dueLine/priorityLine) не попадают в frontmatter.
const tagsYaml     = теги.map(t => "  - " + t).join("\n");
const dueLine      = нуженДедлайн ? 'due_date: "' + дедлайн + '"\n' : "";
const priorityLine = приоритет    ? 'priority: "' + приоритет + '"\n' : "";

// Если предыдущей заметки нет — секции остаются пустыми (как в оригинале).
// Чтобы добавить placeholder, замени "" на нужный текст.
const notesText   = предыдущаяСсылка ? "**Предыдущий материал:** " + предыдущаяСсылка : "";
const relatedText = предыдущаяСсылка ? "- " + предыдущаяСсылка : "";


// ============================================
// БЛОК 9: ФИНАЛИЗАЦИЯ (перемещение файла)
// ============================================
// Templater создаёт заметку во временном месте. tp.file.move()
// переносит её в целевую папку с нужным именем.
// 
// Важно: эта строка должна быть ПОСЛЕДНЕЙ в JS-части шаблона,
// потому что после неё начинается тело заметки (frontmatter + markdown).

await tp.file.move(итоговыйПуть + "/" + имяФайла);

<% tagsYaml %>

<% dueLine %><% priorityLine %>status: "<% статус %>"
